// Status indicator
        if (statusIcon) {
            const canPengajuanModifikasi = {{ auth()->user()->can('pengajuan-modifikasi-laporan') ? 'true' : 'false' }};
            const isModifiable = data.modifiable_by_user_id && data.modifiable_by_user_id == {{ auth()->id() }};

            // Handle the pengajuan data - check if it's an array (multiple requests) or single object
            let pengajuan = data.pengajuan;
            let hasToken = false;

            // If pengajuan is an array (multiple requests), use the most recent one
            if (Array.isArray(pengajuan) && pengajuan.length > 0) {
                // Sort by created_at or updated_at (descending) to get the latest
                const latestPengajuan = pengajuan.sort((a, b) => new Date(b.created_at || b.updated_at) - new Date(a.created_at || a.updated_at))[0];
                hasToken = latestPengajuan && latestPengajuan.status === 'disetujui' && latestPengajuan.token > 0;
            } else if (pengajuan) {
                // If pengajuan is a single object
                hasToken = pengajuan && pengajuan.status === 'disetujui' && pengajuan.token > 0;
            }

            // User can modify if they have direct permission OR they have a valid token for this report
            let isModifiableStatus = canPengajuanModifikasi || (isModifiable && hasToken);

            if (isModifiableStatus) {
                statusIcon.innerHTML = 'Terbuka';
                statusIcon.className = 'badge border-success text-success bg-opacity-20 bg-success fs-7 d-flex align-items-center';
            } else {
                statusIcon.innerHTML = 'Terkunci';
                statusIcon.className = 'badge border-danger text-danger bg-opacity-20 bg-danger fs-7 d-flex align-items-center';
            }

            const ajukanBtn = document.getElementById('ajukanPerubahanBtn');
            if (ajukanBtn) {
                // Show the pengajuan button only if the user cannot modify (no permission and no valid token)
                if (isModifiableStatus) {
                    ajukanBtn.style.display = 'none';
                } else {
                    ajukanBtn.style.display = '';
                }
            }
        }
